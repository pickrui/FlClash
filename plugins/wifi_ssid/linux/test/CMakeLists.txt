cmake_minimum_required(VERSION 3.10)
project(wifi_ssid_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FLUTTER_LINUX_DIR "" CACHE PATH "Flutter Linux engine headers and library directory")
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0 gio-2.0)
find_library(FLUTTER_LIBRARY flutter_linux_gtk PATHS "${FLUTTER_LINUX_DIR}" NO_DEFAULT_PATH)
if(NOT FLUTTER_LIBRARY)
  message(FATAL_ERROR "Set FLUTTER_LINUX_DIR to a Flutter Linux engine directory")
endif()
add_executable(wifi_ssid_plugin_test wifi_ssid_plugin_test.cc)
target_include_directories(wifi_ssid_plugin_test PRIVATE "${FLUTTER_LINUX_DIR}")
target_link_libraries(wifi_ssid_plugin_test PRIVATE PkgConfig::GTK "${FLUTTER_LIBRARY}")
target_compile_options(wifi_ssid_plugin_test PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
enable_testing()
add_test(NAME wifi_ssid_plugin_test COMMAND wifi_ssid_plugin_test)
set_tests_properties(wifi_ssid_plugin_test PROPERTIES TIMEOUT 30)
